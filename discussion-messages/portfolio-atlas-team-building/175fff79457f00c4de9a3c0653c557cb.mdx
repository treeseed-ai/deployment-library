import { lstat, mkdir, realpath, writeFile } from 'node:fs/promises';
import { join } from 'node:path';

export interface SourceCacheFenceOwner {
  jobId: string;
  cacheId: string;
  pid: number;
  operation: 'acquisition' | 'collection';
  assignmentId?: string;
  authorizationId?: string;
}

export async function assertPrivateSourceFence(path: string) {
  const info = await lstat(path);
  if (!info.isDirectory() || info.uid !== process.getuid?.() || (info.mode & 0o077) !== 0 || await realpath(path) !== path) {
    throw new Error('Source cache fence custody is invalid.');
  }
  return info;
}

export async function claimSourceCacheFence(root: string, cacheId: string, owner: SourceCacheFenceOwner) {
  if (!/^[a-f0-9]{64}$/u.test(cacheId) || owner.cacheId !== cacheId
    || !/^[a-f0-9-]{36}$/u.test(owner.jobId) || !Number.isSafeInteger(owner.pid) || owner.pid < 2
    || !['acquisition', 'collection'].includes(owner.operation)) throw new Error('Invalid source cache fence identity.');
  const directory = join(root, 'git-fences');
  await mkdir(directory, { recursive: true, mode: 0o700 });
  await assertPrivateSourceFence(directory);
  const fence = join(directory, `${cacheId}.lock`);
  await mkdir(fence, { mode: 0o700 });
  try {
    await writeFile(join(fence, 'job.json'), JSON.stringify(owner), { mode: 0o600, flag: 'wx' });
    return fence;
  } catch (error) {
    // A fence with incomplete custody is deliberately retained for inspection.
    throw error;
  }
}

  run(command: string, args: string[]): Promise<string>;

const production: CacheRecoveryOperations = {
  alive(pid) {
    try { process.kill(pid, 0); return true; }
    catch (error) { if ((error as NodeJS.ErrnoException).code === 'ESRCH') return false; throw error; }
  },

/** Recover only a trusted-host fetch fence after its owning process is gone.
 * Never kills a process, mounts guest content, or removes source/work/candidate images. */
export async function recoverSourceCacheFence(cache: string, operations = production, stableFence?: string) {
  if (stableFence) await assertPrivateSourceFence(stableFence);
  const lock = join(cache, 'acquisition.lock');
  const exists = await lstat(lock).catch(error => {
    if ((error as NodeJS.ErrnoException).code !== 'ENOENT') throw error;
    return null;
  });
  if (!exists) {
    const cacheInfo = await lstat(cache).catch(error => {
      if ((error as NodeJS.ErrnoException).code !== 'ENOENT') throw error;
      return null;
    });
    if (!cacheInfo) return true;
  }
  if (exists) {
    await owned(cache, true); await owned(lock, true);
    const journal = join(lock, 'job.json'), info = await owned(journal, false);
    if (info.size > 4096) throw new Error('Source cache recovery journal exceeds its bound.');
    const job = JSON.parse(await readFile(journal, 'utf8')) as { pid?: unknown; jobId?: unknown };
    if (!Number.isSafeInteger(job.pid) || Number(job.pid) < 2 || !/^[a-f0-9-]{36}$/u.test(String(job.jobId))) {
      throw new Error('Source cache recovery owner is invalid.');
    }
    if (operations.alive(Number(job.pid))) return false;
  }
  await owned(cache, true);
  const mount = join(cache, 'mounted'), image = join(cache, 'objects.ext4');
  const imageInfo = await lstat(image).catch(error => {
    if ((error as NodeJS.ErrnoException).code !== 'ENOENT') throw error;
    return null;
  });
  if (imageInfo) await owned(image, false);
  const mountInfo = await lstat(mount).catch(error => {
    if ((error as NodeJS.ErrnoException).code !== 'ENOENT') throw error;
    return null;
  });
  if (mountInfo) await owned(mount, true);
  let device = '';
  try { device = await operations.run('/usr/bin/findmnt', ['--noheadings','--output','SOURCE','--mountpoint',mount]); }
  catch (error) { if ((error as { code?: unknown }).code !== 1) throw error; }
  if (device) {
    if (!/^\/dev\/loop[0-9]+$/u.test(device)
      || await operations.run('/usr/sbin/losetup', ['--noheadings','--output','BACK-FILE',device]) !== image) {
      throw new Error('Source cache recovery mount ownership is uncertain.');
    }
    const currentMount = await owned(mount, true);
    const currentImage = await owned(image, false);
    if (!mountInfo || currentMount.dev !== mountInfo.dev || currentMount.ino !== mountInfo.ino
      || !imageInfo || currentImage.dev !== imageInfo.dev || currentImage.ino !== imageInfo.ino) {
      throw new Error('Source cache recovery ownership changed.');
    }
    await operations.run('/usr/bin/umount', [mount]); // busy filesystem remains fenced
    let remainsMounted = '';
    try { remainsMounted = await operations.run('/usr/bin/findmnt', ['--noheadings','--output','SOURCE','--mountpoint',mount]); }
    catch (error) { if ((error as { code?: unknown }).code !== 1) throw error; }
    if (remainsMounted) throw new Error('Source cache recovery mount is still present.');
  }
  if (await operations.run('/usr/sbin/losetup', ['--associated', image])) {
    throw new Error('Source cache recovery still has an attached loop device.');
  }
  const creating = `${image}.creating`;
  const unfinished = await lstat(creating).catch(error => {
    if ((error as NodeJS.ErrnoException).code !== 'ENOENT') throw error;
    return null;
  });
  if (unfinished) { await owned(creating, false); await rm(creating); }
  if (exists) await rm(lock, { recursive: true }); // exact root-owned fetch journal, never execution work
  if (imageInfo) {
    const current = await owned(image, false);
    if (current.dev !== imageInfo.dev || current.ino !== imageInfo.ino) throw new Error('Source cache image ownership changed.');
  }
  return true;
}

/** Recover a stable owner only while it is exclusively marked as being recovered.
 * Acquirers and collectors see the stable directory until every ownership check passes. */
export async function recoverStableSourceCacheFence(root: string, cacheId: string, operations = production) {
  if (!/^[a-f0-9]{64}$/u.test(cacheId)) throw new Error('Invalid source cache fence identity.');
  const fence = join(root, 'git-fences', `${cacheId}.lock`);
  let initial;
  try { initial = await assertPrivateSourceFence(fence); }
  catch (error) { if ((error as NodeJS.ErrnoException).code === 'ENOENT') return true; throw error; }
  const recovery = join(fence, 'recovery.lock');
  try { await mkdir(recovery, { mode: 0o700 }); }
  catch (error) { if ((error as NodeJS.ErrnoException).code === 'EEXIST') return false; throw error; }
  try {
    const names = await readdir(fence);
    if (names.some(name => name !== 'job.json' && name !== 'recovery.lock')) throw new Error('Source cache recovery fence custody is invalid.');
    const journal = join(fence, 'job.json'), info = await owned(journal, false);
    if (info.size > 4096) throw new Error('Source cache recovery journal exceeds its bound.');
    const contents = await readFile(journal, 'utf8');
    const owner = JSON.parse(contents) as { cacheId?: unknown; pid?: unknown; jobId?: unknown; operation?: unknown };
    const record = owner as Record<string, unknown>;
    if (Object.keys(record).some(key => !['jobId', 'cacheId', 'pid', 'operation', 'assignmentId', 'authorizationId'].includes(key))
      || (record.assignmentId !== undefined && typeof record.assignmentId !== 'string')
      || (record.authorizationId !== undefined && typeof record.authorizationId !== 'string')
      || owner.cacheId !== cacheId || !Number.isSafeInteger(owner.pid) || Number(owner.pid) < 2
      || !/^[a-f0-9-]{36}$/u.test(String(owner.jobId)) || !['acquisition', 'collection'].includes(String(owner.operation))) {
      throw new Error('Source cache recovery owner is invalid.');
    }
    if (operations.alive(Number(owner.pid))) return false;
    const cache = join(root, 'git', cacheId);
    const cacheInfo = await lstat(cache).catch(error => { if ((error as NodeJS.ErrnoException).code === 'ENOENT') return undefined; throw error; });
    if (cacheInfo) {
      await recoverSourceCacheFence(cache, operations, fence);
    }
    // Bind deletion to the exact fence directory and journal observed above.
    const current = await assertPrivateSourceFence(fence);
    const currentJournal = await readFile(journal, 'utf8');
    if (current.dev !== initial.dev || current.ino !== initial.ino || currentJournal !== contents) {
      throw new Error('Source cache recovery owner changed.');
    }
    await rm(fence, { recursive: true });
    return true;
  } finally {
    // If the fence still exists, release only our private recovery marker.
    await rm(recovery, { recursive: true }).catch(error => {
      if ((error as NodeJS.ErrnoException).code !== 'ENOENT') throw error;
    });
  }
}
